# IntToFixedPoint has the wrong return type in this beta. A 16-bit model index
# is reconstructed exactly from fixed-point literals, without that conversion.
macro = {
	definition = "CMRgoRoiIndexBit(Index, Divisor, Value)"
	replace_with = "Select_CFixedPoint(IsOdd_int32(Divide_int32(Index, Divisor)), Value, '(CFixedPoint)0')"
}

macro = {
	definition = "CMRgoRoiIndexValue(Index)"
	replace_with = "Add_CFixedPoint(Add_CFixedPoint(Add_CFixedPoint(Add_CFixedPoint(CMRgoRoiIndexBit(Index, '(int32)1', '(CFixedPoint)1'), CMRgoRoiIndexBit(Index, '(int32)2', '(CFixedPoint)2')), Add_CFixedPoint(CMRgoRoiIndexBit(Index, '(int32)4', '(CFixedPoint)4'), CMRgoRoiIndexBit(Index, '(int32)8', '(CFixedPoint)8'))), Add_CFixedPoint(Add_CFixedPoint(CMRgoRoiIndexBit(Index, '(int32)16', '(CFixedPoint)16'), CMRgoRoiIndexBit(Index, '(int32)32', '(CFixedPoint)32')), Add_CFixedPoint(CMRgoRoiIndexBit(Index, '(int32)64', '(CFixedPoint)64'), CMRgoRoiIndexBit(Index, '(int32)128', '(CFixedPoint)128')))), Add_CFixedPoint(Add_CFixedPoint(Add_CFixedPoint(CMRgoRoiIndexBit(Index, '(int32)256', '(CFixedPoint)256'), CMRgoRoiIndexBit(Index, '(int32)512', '(CFixedPoint)512')), Add_CFixedPoint(CMRgoRoiIndexBit(Index, '(int32)1024', '(CFixedPoint)1024'), CMRgoRoiIndexBit(Index, '(int32)2048', '(CFixedPoint)2048'))), Add_CFixedPoint(Add_CFixedPoint(CMRgoRoiIndexBit(Index, '(int32)4096', '(CFixedPoint)4096'), CMRgoRoiIndexBit(Index, '(int32)8192', '(CFixedPoint)8192')), Add_CFixedPoint(CMRgoRoiIndexBit(Index, '(int32)16384', '(CFixedPoint)16384'), CMRgoRoiIndexBit(Index, '(int32)32768', '(CFixedPoint)32768')))))"
}

	definition = "CMRgoRoiSourceIndex"

}
